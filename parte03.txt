const FECHA_INICIO_APOD = "1995-06-16";
const FECHA_MINIMA = "1900-01-01";
const NOMBRES_MESES = [
"enero", "febrero", "marzo", "abril", "mayo", "junio",
"julio", "agosto", "septiembre", "octubre", "noviembre", "diciembre",
];
function construirFechaTexto(dia, mes, anio) {
const d = Number(dia);
const m = Number(mes);
const a = Number(anio);
if (!d || !m || !a) return "";
const dosCifras = (n) => String(n).padStart(2, "0");
return `${String(a).padStart(4, "0")}-${dosCifras(m)}-${dosCifras(d)}`;
}
function obtenerPartesFecha(fechaTexto) {
const coincidencia = /^(\d{4})-(\d{2})-(\d{2})$/.exec(fechaTexto || "");
if (!coincidencia) return null;
const anio = Number(coincidencia[1]);
const mes = Number(coincidencia[2]);
const dia = Number(coincidencia[3]);
const diasDelMes = new Date(anio, mes, 0).getDate();
if (mes < 1 || mes > 12 || dia < 1 || dia > diasDelMes) return null;
return { dia, mes, anio };
}
function obtenerHoyTexto() {
const hoy = new Date();
return construirFechaTexto(hoy.getDate(), hoy.getMonth() + 1, hoy.getFullYear());
}
const NOMBRES_DIAS = ["lunes", "martes", "miércoles", "jueves", "viernes", "sábado", "domingo"];
function obtenerDiaSemana({ dia, mes, anio }) {
const diaJs = new Date(anio, mes - 1, dia).getDay(); // JavaScript usa 0 = domingo … 6 = sábado
const numero = diaJs === 0 ? 7 : diaJs;
return { numero, nombre: NOMBRES_DIAS[numero - 1] };
}
const ANCHO_BITS = { dia: 5, mes: 4, anio: 11 };
function obtenerBitsFecha({ dia, mes, anio }) {
return {
dia: dia.toString(2).padStart(ANCHO_BITS.dia, "0"),
mes: mes.toString(2).padStart(ANCHO_BITS.mes, "0"),
anio: anio.toString(2).padStart(ANCHO_BITS.anio, "0"),
};
}
function obtenerDiaAnteriorTexto(fechaTexto) {
const { dia, mes, anio } = obtenerPartesFecha(fechaTexto);
const anterior = new Date(anio, mes - 1, dia - 1);
return construirFechaTexto(anterior.getDate(), anterior.getMonth() + 1, anterior.getFullYear());
}
function validarFecha(fechaTexto) {
if (!fechaTexto) {
return { valida: false, mensaje: "Completa el día, el mes y el año para ver tu cielo." };
}
const partes = obtenerPartesFecha(fechaTexto);
if (!partes) {
return { valida: false, mensaje: "Esa fecha no existe. Revisa el día y el mes." };
}
if (partes.anio < 1000) {
return { valida: false, mensaje: "Escribe el año completo, con cuatro cifras (por ejemplo, 1998)." };
}
if (fechaTexto < FECHA_MINIMA) {
return { valida: false, mensaje: "Prueba con una fecha a partir de 1900." };
}
if (fechaTexto > obtenerHoyTexto()) {
return { valida: false, mensaje: "Esa fecha todavía no ha llegado. Elige hoy o un día anterior." };
}
return { valida: true, mensaje: "" };
}
function formatearFechaLarga({ dia, mes, anio }) {
return `${dia} de ${NOMBRES_MESES[mes - 1]} de ${anio}`;
}
