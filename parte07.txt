function describirBrillo(magnitud) {
if (magnitud < 1.5) return "de las más brillantes del cielo";
if (magnitud < 2.5) return "brillante";
if (magnitud < 3.5) return "se ve a simple vista";
return "tenue";
}
function obtenerEstrellasConNombre(constelacion) {
return constelacion.estrellas
.filter((estrella) => estrella.nombre)
.sort((a, b) => a.mag - b.mag)
.map((estrella) => ({ nombre: estrella.nombre, mag: estrella.mag, brillo: describirBrillo(estrella.mag) }));
}
function crearListaEstrellas(constelacion, { conBrillo = true } = {}) {
return obtenerEstrellasConNombre(constelacion).map(({ nombre, mag, brillo }) => {
const item = document.createElement("li");
const punto = document.createElement("span");
punto.className = "punto-estrella";
punto.style.setProperty("--tamano", `${Math.max(0.35, 1 - mag * 0.17).toFixed(2)}rem`);
const texto = document.createElement("span");
const fuerte = document.createElement("strong");
fuerte.textContent = nombre;
texto.append(fuerte);
if (conBrillo) texto.append(` · ${brillo}`);
item.append(punto, texto);
return item;
});
}
