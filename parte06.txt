const mediodia = new Date(anio, mes - 1, dia, 12, 0, 0);

}
function calcularLongitudSol(fechaJuliana) {
const n = fechaJuliana - 2451545.0; // días desde el 1 de enero de 2000
const longitudMedia = 280.46 + 0.9856474 * n;
const anomalia = (357.528 + 0.9856003 * n) * GRADOS;
const longitud = longitudMedia + 1.915 * Math.sin(anomalia) + 0.02 * Math.sin(2 * anomalia);
const longitudJ2000 = longitud - 1.397 * (n / 36525);
return ((longitudJ2000 % 360) + 360) % 360;
}

function calcularFaseLunar(partes) {
