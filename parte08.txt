const ZONAS_BITS = {
dia: { x0: 25, x1: 290, y0: 62, y1: 250, nombre: "Día", arriba: true },
mes: { x0: 320, x1: 585, y0: 62, y1: 230, nombre: "Mes", arriba: true },
anio: { x0: 20, x1: 585, y0: 318, y1: 530, nombre: "Año" },
};
function posicionesGrupoBits(clave, n, semilla) {
const { x0, x1, y0, y1 } = ZONAS_BITS[clave];
const azar = crearAzar(`${semilla}-${clave}`);
const paso = (x1 - x0) / n;
let y = y0 + azar() * (y1 - y0);
return Array.from({ length: n }, (_, i) => {
const x = x0 + (i + 0.5) * paso + (azar() - 0.5) * paso * 0.55;
if (i > 0) y = Math.min(y1, Math.max(y0, y + (azar() - 0.5) * (y1 - y0) * 0.9));
return [x, y];
});
}
