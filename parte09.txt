capaLineas.append(linea);
}
const radio = 6 + (5 * (campo.length - 1 - i)) / Math.max(1, campo.length - 1);
const brillo = crearSvg("circle", { cx: x, cy: y, r: radio * 3.4, fill: "url(#halo-bits)" });
brillo.style.setProperty("--orden", ordenEstrella);
capaHalos.append(brillo);
const punto = crearSvg("circle", { cx: x, cy: y, r: radio });
punto.style.setProperty("--orden", ordenEstrella++);
capaEstrellas.append(punto);
anterior = [x, y];
encendidas[clave].push([x, y]);
} else {
capaApagadas.append(crearSvg("circle", { cx: x, cy: y, r: 5 }));
}
if (conNombres) {
const digito = crearSvg("text", { x, y: y + 30, "text-anchor": "middle", class: "digito" });
digito.textContent = bit;
capaNombres.append(digito);
}
});
if (conNombres) {
const { nombre, x0, x1, arriba } = ZONAS_BITS[clave];
const ys = posiciones.map(([, py]) => py);
const yEtiqueta = arriba ? Math.min(...ys) - 26 : Math.max(...ys) + 58;
const etiqueta = crearSvg("text", { x: (x0 + x1) / 2, y: yEtiqueta, "text-anchor": "middle" });
etiqueta.textContent = `${nombre} ${partes[clave]}`;
capaNombres.append(etiqueta);
}
}
for (const clave of ["dia", "mes"]) {
let mejor = null;
for (const a of encendidas[clave]) {
for (const b of encendidas.anio) {
const d = Math.hypot(a[0] - b[0], a[1] - b[1]);
if (!mejor || d < mejor.d) mejor = { a, b, d };
}
}
if (mejor) capaPuentes.append(crearSvg("line", { x1: mejor.a[0], y1: mejor.a[1], x2: mejor.b[0], y2: mejor.b[1] }));
}
svg.append(defs, crearSvg("g", { class: "foto" }), capaFondo, capaApagadas, capaPuentes, capaLineas, capaHalos, capaEstrellas, capaNombres);
return svg;
}
function dibujarLuna(fraccion) {
const r = 50;
const svg = crearSvg("svg", { viewBox: "0 0 100 100" });
svg.append(crearSvg("circle", { cx: r, cy: r, r, fill: "#1d2244" }));
const creciente = fraccion < 0.5;
const ladoBorde = creciente ? 1 : 0;
const anchoTerminador = r * Math.abs(Math.cos(2 * Math.PI * fraccion));
const curvaTerminador = (fraccion % 0.5) < 0.25 ? 0 : 1;
const parteIluminada = crearSvg("path", {
d: `M ${r} 0 A ${r} ${r} 0 0 ${ladoBorde} ${r} ${2 * r} ` +
`A ${anchoTerminador} ${r} 0 0 ${curvaTerminador} ${r} 0 Z`,
fill: "#f3ecd2",
});
if (HEMISFERIO_SUR) parteIluminada.setAttribute("transform", "translate(100 0) scale(-1 1)");
svg.append(parteIluminada);
return svg;
}
